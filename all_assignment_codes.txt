# Automatic differentiation using GradientTape
x = tf.Variable(3.0)

    y = x ** 2 + 2 * x

# Calculate derivative
gradient = tape.gradient(y, x)

print("dy/dx of x^2 + 2x at x=3:", gradient.numpy())

# Experiment 4: Autoencoder for Anomaly Detection (MNIST)
# Digit 1 = Normal, Other digits = Anomalies

# 1. Import Libraries
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers
from sklearn.metrics import (
    classification_report,
    confusion_matrix,
    roc_auc_score
)
from tensorflow.keras.datasets import mnist

# Set random seeds
tf.random.set_seed(42)
np.random.seed(42)

# 2. Load MNIST Dataset
(x_train, y_train), (x_test, y_test) = mnist.load_data()

# 3. Preprocess Dataset
x_train = x_train.reshape(-1, 784).astype("float32") / 255.0
x_test = x_test.reshape(-1, 784).astype("float32") / 255.0

NORMAL = 1

# Train only on normal digit images
x_normal = x_train[y_train == NORMAL]

# Select test samples
rng = np.random.default_rng(42)

norm_idx = np.where(y_test == NORMAL)[0]

anom_idx = rng.choice(
    np.where(y_test != NORMAL)[0],
    size=150,
    replace=False
)

idx = np.concatenate([norm_idx, anom_idx])

x_eval = x_test[idx]
y_eval = (y_test[idx] != NORMAL).astype(int)

print("Normal training samples:", x_normal.shape[0])

print(
    "Test samples:", len(idx),
    "(normal:", (y_eval == 0).sum(),
    "anomalies:", (y_eval == 1).sum(), ")"
)

# 4. Build Autoencoder Model

# Encoder
inp = keras.Input((784,))

e = layers.Dense(128, activation="relu")(inp)
e = layers.Dense(64, activation="relu")(e)

code = layers.Dense(
    16,
    activation="relu",
    name="latent"
)(e)

# Decoder
d = layers.Dense(64, activation="relu")(code)
d = layers.Dense(128, activation="relu")(d)

out = layers.Dense(784, activation="sigmoid")(d)

autoencoder = keras.Model(
    inp,
    out,
    name="autoencoder"
)

# Display Model Summary
autoencoder.summary()

# 5. Compile Autoencoder
autoencoder.compile(
    optimizer="adam",
    loss="mse",
    metrics=["mae"]
)

# 6. Train Autoencoder
H = autoencoder.fit(
    x_normal,
    x_normal,
    epochs=30,
    batch_size=128,
    validation_split=0.1,
    shuffle=True,
    verbose=2
)

# 7. Calculate Reconstruction Error
def rec_error(x):
    reconstructed = autoencoder.predict(x, verbose=0)
    return np.mean((x - reconstructed) ** 2, axis=1)

# Calculate training reconstruction error
train_err = rec_error(x_normal)

# Set anomaly threshold
threshold = train_err.mean() + 2 * train_err.std()

# Calculate test reconstruction error
err = rec_error(x_eval)

# Predict anomalies
y_pred = (err > threshold).astype(int)

# 8. Display Evaluation Results
print(
    f"\nThreshold (mean + 2*std of training error): "
    f"{threshold:.5f}"
)

print(
    f"Mean reconstruction error - normal: "
    f"{err[y_eval == 0].mean():.5f} | "
    f"anomaly: {err[y_eval == 1].mean():.5f}"
)

print("ROC-AUC:", round(roc_auc_score(y_eval, err), 4))

print(
    "Confusion matrix [[TN FP],[FN TP]]:\n",
    confusion_matrix(y_eval, y_pred)
)

print("\nClassification Report:")

print(
    classification_report(
        y_eval,
        y_pred,
        target_names=["normal", "anomaly"],
        digits=4
    )
)

# 9. Plot Exactly Two Graphs
plt.style.use("default")

fig, ax = plt.subplots(1, 2, figsize=(11, 4))

# Graph 1: Autoencoder Training Loss
ax[0].plot(H.history["loss"], label="train_loss")
ax[0].plot(H.history["val_loss"], label="val_loss")

ax[0].set_title("Autoencoder training loss (MSE)")
ax[0].set_xlabel("Epoch")
ax[0].legend()
ax[0].grid(alpha=0.3)

# Graph 2: Reconstruction Error
ax[1].hist(
    err[y_eval == 0],
    bins=50,
    alpha=0.7,
    label="normal"
)

ax[1].hist(
    err[y_eval == 1],
    bins=50,
    alpha=0.7,
    label="anomaly"
)

ax[1].axvline(
    threshold,
    color="k",
    linestyle="--",
    label="threshold"
)

ax[1].set_title("Reconstruction error")
ax[1].set_xlabel("MSE")
ax[1].legend()
ax[1].grid(alpha=0.3)

plt.tight_layout()
plt.savefig("a4_error.png", dpi=130)
plt.show()

# 10. Display Original and Reconstructed Images

sel = (
    list(np.where(y_eval == 0)[0][:5]) +
    list(np.where(y_eval == 1)[0][:5])
)

rec = autoencoder.predict(x_eval[sel], verbose=0)

fig, ax = plt.subplots(2, 10, figsize=(12, 2.8))

for i, k in enumerate(sel):

    # Original images
    ax[0, i].imshow(
        x_eval[k].reshape(28, 28),
        cmap="gray"
    )
    ax[0, i].axis("off")

    # Reconstructed images
    ax[1, i].imshow(
        rec[i].reshape(28, 28),
        cmap="gray"
    )
    ax[1, i].axis("off")

    ax[0, i].set_title(
        "normal" if y_eval[k] == 0 else "anomaly",
        fontsize=8
    )

ax[0, 0].text(
    -0.3, 0.5, "input",
    transform=ax[0, 0].transAxes,
    rotation=90,
    va="center",
    ha="right",
    fontsize=8
)

ax[1, 0].text(
    -0.3, 0.5, "output",
    transform=ax[1, 0].transAxes,
    rotation=90,
    va="center",
    ha="right",
    fontsize=8
)

plt.tight_layout()
plt.savefig("a4_recon.png", dpi=130)
plt.show()

# 11. Conclusion
print("\nAutoencoder for Anomaly Detection implemented successfully!")

# Assignment 5 - Continuous Bag of Words (CBOW) with Gensim Word2Vec
import re, nltk, warnings
warnings.filterwarnings("ignore")
import gensim
from gensim.models import Word2Vec
from nltk.corpus import stopwords
from nltk.tokenize import sent_tokenize, word_tokenize
# Step 1: dataset - "Alice's Adventures in Wonderland" (Project Gutenberg #11, same text as
# 11-0.txt)
text = nltk.corpus.gutenberg.raw("carroll-alice.txt")
print("Characters in raw text:", len(text))
# Steps 2-6: pre-processing
sentences = sent_tokenize(text) # Step 5 (sentences)
stop_words = set(stopwords.words("english"))
data = []
for s in sentences:
 # Steps 2-3: remove special characters, digits, punctuation
 s = re.sub(r"[^A-Za-z\s]", " ", s)
 s = s.lower() # Step 4: lower case
 words = [w for w in word_tokenize(s) if w not in stop_words and len(w) > 1] # Steps 5-6
 if words: data.append(words)
print("Sentences after cleaning:", len(data))
print("Sample sentence:", data[10][:10])
# Step 7: train CBOW model (sg=0 -> CBOW, sg=1 -> Skip-gram)
model = Word2Vec(sentences=data, vector_size=100, window=5, min_count=2, sg=0, workers=1,
 epochs=200, seed=1)
print("Gensim version:", gensim.__version__)
print("Vocabulary size:", len(model.wv))
print("Vector for 'alice' (first 8 dims):", model.wv["alice"][:8].round(4))
# Steps 8-9: similarity and prediction
print("\nSimilarity('alice','queen'):", round(float(model.wv.similarity("alice", "queen")), 4))
print("Similarity('alice','rabbit'):", round(float(model.wv.similarity("alice", "rabbit")), 4))
for w in ["alice", "queen", "king", "rabbit"]:
 print(f"\nTop-5 words most similar to '{w}':")
 for word, sc in model.wv.most_similar(w, topn=5): print(f" {word:<12}{sc:.4f}")
print("\nPredict output word for context ['mad','tea','party'] :")
for word, p in model.predict_output_word(["mad", "tea", "party"],
 topn=5): print(f" {word:<12}{p:.4f}")
print("Predict output word for context ['white','rabbit','hole']:")
for word, p in model.predict_output_word(["white", "rabbit", "hole"],
 topn=5): print(f" {word:<12}{p:.4f}")
